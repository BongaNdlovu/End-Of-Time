rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    // This rule is required for collectionGroup queries on 'interactions'
    // It allows any client to read interaction documents (e.g., to count prayers).
    match /{path=**}/interactions/{interactionId} {
      allow read: if true;
      // Keep the more specific create/update/delete rules below for security.
    }

    // Leaderboard collection rules - Enhanced with data validation
    match /leaderboard/{userId} {
      // Anyone can read leaderboard entries (for displaying the leaderboard)
      allow read: if true;

      // Users can only write to their own document with strict validation
      allow create, update: if request.auth != null
        && request.auth.uid == userId
        // Validate required fields exist
        && request.resource.data.keys().hasAll(['score', 'level', 'name', 'date'])
        // Validate data types and ranges
        && request.resource.data.score is int
        && request.resource.data.score >= 0
        && request.resource.data.score <= 50000  // Reasonable max score
        && request.resource.data.level is int
        && request.resource.data.level >= 1
        && request.resource.data.level <= 7
        && request.resource.data.name is string
        && request.resource.data.name.size() >= 1
        && request.resource.data.name.size() <= 50
        // Validate correct answers and question count if provided
        && (!request.resource.data.keys().hasAny(['correctAnswers']) ||
            (request.resource.data.correctAnswers is int && request.resource.data.correctAnswers >= 0))
        && (!request.resource.data.keys().hasAny(['questionCount']) ||
            (request.resource.data.questionCount is int && request.resource.data.questionCount >= 0))
        // Rate limiting: prevent rapid submissions (max 1 per 30 seconds)
        && (!resource.exists() || request.time > resource.data.date + duration.value(30, 's'));

      // Prevent deletion of leaderboard entries (preserve history)
      allow delete: if false;
    }
    
    // Test collection for validation (optional)
    match /test/{document} {
      allow read: if true;
      allow write: if request.auth != null;
    }

    // Prayers collection rules
    match /prayers/{prayerId} {
      // Publicly readable feed (client filters anonymity/ownership)
      allow read: if true;

      // Only authenticated users can create prayer requests with validation
      allow create: if request.auth != null
        && request.resource.data.userId == request.auth.uid
        && request.resource.data.keys().hasAll(['userId', 'timestamp'])
        // Prayer content validation
        && (!request.resource.data.keys().hasAny(['content']) ||
            (request.resource.data.content is string &&
             request.resource.data.content.size() >= 1 &&
             request.resource.data.content.size() <= 2000))
        // Timestamp must be recent
        && request.resource.data.timestamp > request.time - duration.value(5, 's')
        && request.resource.data.timestamp <= request.time;

      // Only the owner can update/delete their own prayer
      allow update, delete: if request.auth != null && request.auth.uid == resource.data.userId;
    }

    // Interactions on a prayer (e.g., "I prayed", comments)
    match /prayers/{prayerId}/interactions/{interactionId} {
      allow read: if true;

      // Allow interaction creation with validation (supports authenticated and guest users)
      allow create: if request.resource.data.keys().hasAll(['type', 'timestamp'])
        && request.resource.data.type in ['prayed', 'comment']
        // Timestamp must be recent (within last 5 seconds)
        && request.resource.data.timestamp > request.time - duration.value(5, 's')
        && request.resource.data.timestamp <= request.time
        // If comment type, must have content
        && (request.resource.data.type != 'comment' ||
            (request.resource.data.keys().hasAny(['content']) &&
             request.resource.data.content is string &&
             request.resource.data.content.size() > 0 &&
             request.resource.data.content.size() <= 500))
        // If authenticated, userId must match auth
        && (!request.resource.data.keys().hasAny(['userId']) ||
            (request.auth != null && request.resource.data.userId == request.auth.uid));

      // Only the author of an interaction can modify/delete it
      allow update, delete: if request.auth != null && request.auth.uid == resource.data.userId;
    }

    // Notifications for users
    match /notifications/{notificationId} {
      allow read: if request.auth != null && request.auth.uid == resource.data.userId;

      // Restrict notification creation to authenticated users only
      allow create: if request.auth != null
        && request.resource.data.userId == request.auth.uid
        && request.resource.data.keys().hasAll(['userId', 'timestamp'])
        // Validate notification content
        && (!request.resource.data.keys().hasAny(['message']) ||
            (request.resource.data.message is string &&
             request.resource.data.message.size() >= 1 &&
             request.resource.data.message.size() <= 500));

      allow update, delete: if request.auth != null && request.auth.uid == resource.data.userId;
    }

    // User profiles (store FCM tokens and preferences)
    match /users/{userId} {
      allow read, write: if request.auth != null && request.auth.uid == userId;
    }
  }
}
