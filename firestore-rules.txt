rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    // Helper to ensure requests include a valid App Check token
    function hasValidAppCheck() {
      return request.app == null || request.app.token == null || request.app.token.valid;
    }

    // ============================================
    // GAME COLLECTIONS (Used by trivia game)
    // ============================================

    // Leaderboard collection rules
    match /leaderboard/{userId} {
      // Anyone can read leaderboard entries (for displaying the leaderboard)
      allow read: if true;
      // Users can create/update their own document only
      allow create, update: if request.auth != null && request.auth.uid == userId;
      // Disallow deletes to preserve leaderboard integrity
      allow delete: if false;
    }

    // ============================================
    // Prayer Network collections have been removed
    // ============================================

    // User profiles
    match /users/{userId} {
      allow read, write: if request.auth != null
                          && hasValidAppCheck()
                          && request.auth.uid == userId;
    }

    // ============================================
    // PRESENTATION/ACADEMY COLLECTIONS
    // ============================================

    // Comments collection for presentations
    match /comments/{commentId} {
      // Anyone can read comments (public visibility)
      allow read: if true;

      // Authenticated users (including anonymous) can create comments with validation
      allow create: if request.auth != null
        && request.resource.data.text is string
        && request.resource.data.text.size() > 0
        && request.resource.data.text.size() <= 1000
        && request.resource.data.author is string
        && request.resource.data.author.size() > 0
        && request.resource.data.author.size() <= 100
        && request.resource.data.presentationId is string
        && request.resource.data.presentationId.matches('presentation-[0-9]+')
        && (request.resource.data.userId == request.auth.uid || request.resource.data.userId == 'anonymous');

      // Users can update/delete only their own comments
      allow update, delete: if request.auth != null
        && request.auth.uid == resource.data.userId;
    }

    // Quiz completions collection
    match /quiz_completions/{completionId} {
      // Anyone can read aggregated stats (for analytics)
      allow read: if true;

      // Authenticated users can create their own completion records
      allow create: if request.auth != null
        && request.resource.data.userId == request.auth.uid
        && request.resource.data.score is int
        && request.resource.data.score >= 0
        && request.resource.data.total is int
        && request.resource.data.total > 0
        && request.resource.data.score <= request.resource.data.total
        && request.resource.data.presentationId is string
        && request.resource.data.timestamp == request.time;

      // No updates or deletes allowed (preserve historical data)
      allow update, delete: if false;
    }

    // Test collection for validation (optional - can be removed in production)
    match /test/{document} {
      allow read: if true;
      allow write: if request.auth != null;
    }
  }
}
